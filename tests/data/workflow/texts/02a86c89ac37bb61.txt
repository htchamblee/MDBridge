set -eo pipefail
cd "$(dirname "$0")"

# MDBridge: the mdbridge program on PATH, unless MDBRIDGE names another
MDBRIDGE="${MDBRIDGE:-mdbridge}"

# Prepare the starting structure for tleap (hydrogens, 5'-phosphates, TER records)
python3 golden_03_prep_20261015-1432.py /golden/inputs/structure.pdb golden_03_prep_20261015-1432.pdb

# L-RNA residue names from the sugar chirality (C1'/C3'/C4'): LA/LC/LG/LU, LA5/LA3 ...
"$MDBRIDGE" pdb clean golden_03_prep_20261015-1432.pdb -o golden_03_lrna_20261015-1432.pdb --scheme amber
[ -s "golden_03_lrna_20261015-1432.pdb" ] || { echo "ERROR: mdbridge pdb clean did not write golden_03_lrna_20261015-1432.pdb" >&2; exit 1; }

# Solvate and add ions
rm -f golden_03_tleap_20261015-1432.parm7 golden_03_tleap_20261015-1432.rst7
tleap -f golden_03_tleap_20261015-1432.leap | tee golden_03_tleap_20261015-1432.log
[ -s "golden_03_tleap_20261015-1432.parm7" ] || { echo "ERROR: tleap did not write golden_03_tleap_20261015-1432.parm7" >&2; exit 1; }

# Apply NBfix19 + 0BPh
parmed -O golden_03_tleap_20261015-1432.parm7 -i golden_03_parmed_20261015-1432.in 2>&1 | tee golden_03_parmed_20261015-1432.log
[ -s "golden_03_parmed_20261015-1432.parm7" ] || { echo "ERROR: ParmEd did not write golden_03_parmed_20261015-1432.parm7" >&2; exit 1; }

echo "Topology: golden_03_parmed_20261015-1432.parm7  Coordinates: golden_03_tleap_20261015-1432.rst7"
