#!/bin/bash
# RNAMake design_rna_scaffold
# Project apt1, step 08 - written by MDBridge <VERSION> on <DATE>.
# MDBridge only writes this script; review it, then run it with: bash apt1_08_rnamake_20261015-1432.sh
# Please cite (RNAMake):
#   - Yesselman J. D., Eiler D., Carlson E. D., Gotrik M. R., d'Aquino A. E., Ooms A. N., Kladwang
#     W., Carlson P. D., Shi X., Costantino D. A., et al. Nat. Nanotechnol. 2019, 14, 866-873.
#     doi:10.1038/s41565-019-0517-8

set -eo pipefail
cd "$(dirname "$0")"

export RNAMAKE="${RNAMAKE:?set RNAMAKE to the RNAMake folder}"

# The design PDBs of an earlier run go first
rm -f design.*.pdb apt1_08_design-[0-9]*_20261015-1432.pdb

# Build motif paths between the base pairs, optimise sequences, score designs
design_rna_scaffold \
  --pdb ../07_simrna/apt1_07_model-01_20261015-1432.pdb \
  --start_bp A1-A12 \
  --end_bp A6-A7 \
  --designs 10 \
  --out_file apt1_08_design_20261015-1432.out \
  --score_file apt1_08_design_20261015-1432.scores \
  --search_cutoff 7.5 \
  --search_max_size 999999 \
  --search_max_motifs 999 \
  --min_helix_length 4 --max_helix_length 99 \
  --sequences_per_design 1 \
  --seq_opt_steps 10000 --seq_opt_cutoff 7.0 \
  --dump_pdbs > apt1_08_design_20261015-1432.log 2>&1
[ -s "apt1_08_design_20261015-1432.scores" ] || { echo "ERROR: design_rna_scaffold did not write apt1_08_design_20261015-1432.scores" >&2; exit 1; }

# Number the design PDBs (RNAMake writes design.0.pdb, design.1.pdb ...)
for f in design.*.pdb; do
  [ -e "$f" ] || continue
  K=${f#design.}; K=${K%.pdb}
  N=$(printf "%02d" $((K + 1)))
  mv "$f" "apt1_08_design-${N}_20261015-1432.pdb"
done
