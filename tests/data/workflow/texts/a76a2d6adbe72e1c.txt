#!/bin/bash
# MDBridge structure builder: Rigid-body fit into a density map
# Project golden, step 03 - written by MDBridge <VERSION> on <DATE>.
# MDBridge only writes this script; review it, then run it with: bash golden_03_build_20261015-1432.sh
# MDBridge must be on PATH, or MDBRIDGE must name the program.

set -eo pipefail
cd "$(dirname "$0")"

# MDBridge: the mdbridge program on PATH, unless MDBRIDGE names another
MDBRIDGE="${MDBRIDGE:-mdbridge}"

"$MDBRIDGE" build fit /golden/inputs/pdb.pdb /golden/inputs/map.map --resolution 4 --search global --step 30 -o golden_03_fitted_20261015-1432.pdb
[ -s "golden_03_fitted_20261015-1432.pdb" ] || { echo "ERROR: mdbridge build did not write golden_03_fitted_20261015-1432.pdb" >&2; exit 1; }
