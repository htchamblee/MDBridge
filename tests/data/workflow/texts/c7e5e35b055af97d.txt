#!/bin/bash
# FARFAR2 models: cluster, extract, annotate
# Project golden, step 03 - written by MDBridge <VERSION> on <DATE>.
# MDBridge only writes this script; review it, then run it with: bash golden_03_farfar2-models_20261015-1432.sh
# Please cite (Rosetta, FARFAR2, DSSR (x3dna-dssr)):
#   - Leman J. K., Weitzner B. D., Lewis S. M., Adolf-Bryfogle J., Alam N., Alford R. F., Aprahamian
#     M., Baker D., Barlow K. A., Barth P., et al. Nat. Methods 2020, 17, 665-680.
#     doi:10.1038/s41592-020-0848-2
#   - Watkins A. M., Rangan R., Das R. Structure 2020, 28, 963-976. doi:10.1016/j.str.2020.05.011
#   - Lu X.-J., Bussemaker H. J., Olson W. K. Nucleic Acids Res. 2015, 43, e142.
#     doi:10.1093/nar/gkv716

set -eo pipefail
cd "$(dirname "$0")"

# rank_models FILE: prints 'score tag file' for every model in a silent file
rank_models() {
  awk -v f="$1" '$1=="SCORE:" { if ($0 ~ /description/) { s=2; d=NF; for (i=2; i<=NF; i++) { if ($i=="score") s=i; if ($i=="description") d=i }; next } print $s, $d, f }' "$1"
}

# Cluster all models (best cluster first)
rm -f golden_03_cluster_20261015-1432.out
rna_cluster -in:file:silent /golden/inputs/silent.silent -out:file:silent golden_03_cluster_20261015-1432.out -cluster:radius 2.0 -nstruct 400 > golden_03_cluster_20261015-1432.log 2>&1
[ -s "golden_03_cluster_20261015-1432.out" ] || { echo "ERROR: rna_cluster did not write golden_03_cluster_20261015-1432.out" >&2; exit 1; }
rank_models golden_03_cluster_20261015-1432.out > golden_03_ranking_20261015-1432.txt

# Extract the best 10 models as numbered PDB files (those of an earlier run go first)
rm -f golden_03_models_20261015-1432.txt golden_03_model-[0-9]*_20261015-1432.pdb
K=0
while read -r SCORE TAG FILE; do
  K=$((K + 1))
  [ $K -gt 10 ] && break
  N=$(printf "%02d" "$K")
  rm -f "$TAG.pdb"
  extract_pdbs -in:file:silent "$FILE" -in:file:tags "$TAG" > /dev/null || true
  if [ -s "$TAG.pdb" ]; then mv "$TAG.pdb" "golden_03_model-${N}_20261015-1432.pdb"; echo "$N $SCORE $TAG $FILE" >> golden_03_models_20261015-1432.txt; else echo "could not extract $TAG from $FILE" >&2; fi
done < golden_03_ranking_20261015-1432.txt

# Annotate every model with DSSR
for f in golden_03_model-[0-9]*_20261015-1432.pdb; do
  [ -e "$f" ] || continue
  x3dna-dssr -i="$f" -o="${f%.pdb}.dssr"
  x3dna-dssr -i="$f" --json -o="${f%.pdb}.json"
done
x3dna-dssr --cleanup > /dev/null 2>&1 || true

[ -s "golden_03_model-01_20261015-1432.pdb" ] || { echo "ERROR: the extraction did not write golden_03_model-01_20261015-1432.pdb" >&2; exit 1; }
echo "Best model: golden_03_model-01_20261015-1432.pdb (ranking in golden_03_models_20261015-1432.txt)"
