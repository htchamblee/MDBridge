#!/bin/bash
# Restraint dictionaries (acedrg) for 0C3 0G5
# Project golden, step 03 - written by MDBridge <VERSION> on <DATE>.
# MDBridge only writes this script; review it, then run it with: bash golden_03_dicts_20261015-1432.sh
# Please cite (AceDRG (CCP4), wwPDB Chemical Component Dictionary):
#   - Long F., Nicholls R. A., Emsley P., Gražulis S., Merkys A., Vaitkus A., Murshudov G. N. Acta
#     Cryst. D 2017, 73, 112-122. doi:10.1107/S2059798317000067
#   - Agirre J., et al. Acta Cryst. D 2023, 79, 449-461. doi:10.1107/S2059798323003595
#   - Westbrook J. D., Shao C., Feng Z., Zhuravleva M., Velankar S., Young J. Bioinformatics 2015,
#     31, 1274-1278. doi:10.1093/bioinformatics/btu789

set -eo pipefail
cd "$(dirname "$0")"

# MDBridge: the mdbridge program on PATH, unless MDBRIDGE names another
MDBRIDGE="${MDBRIDGE:-mdbridge}"

command -v acedrg >/dev/null || { echo "ERROR: acedrg not found on PATH" >&2; exit 1; }

# 0C3: L-CYTIDINE-5'-MONOPHOSPHATE (3'-TERMINAL, NO OP3) - SMILES from the MDATK table
# generated with the leaving atom OP3; 'finish' removes it
rm -rf work_0C3 && mkdir -p work_0C3
( cd work_0C3 && acedrg -i ../golden_03_0C3_20261015-1432.smi -r 0C3 -o 0C3 ) > work_0C3/acedrg.log 2>&1 \
    || { echo "ERROR: acedrg failed for 0C3 (see work_0C3/acedrg.log)" >&2; exit 1; }
"$MDBRIDGE" dict finish work_0C3 --code 0C3 --group RNA -o golden_03_0C3_20261015-1432.cif
[ -s "golden_03_0C3_20261015-1432.cif" ] || { echo "ERROR: mdbridge dict finish did not write golden_03_0C3_20261015-1432.cif" >&2; exit 1; }

# 0G5: L-GUANOSINE (5'-TERMINAL, 5'-OH) - SMILES from the MDATK table
rm -rf work_0G5 && mkdir -p work_0G5
( cd work_0G5 && acedrg -i ../golden_03_0G5_20261015-1432.smi -r 0G5 -o 0G5 ) > work_0G5/acedrg.log 2>&1 \
    || { echo "ERROR: acedrg failed for 0G5 (see work_0G5/acedrg.log)" >&2; exit 1; }
"$MDBRIDGE" dict finish work_0G5 --code 0G5 --group RNA -o golden_03_0G5_20261015-1432.cif
[ -s "golden_03_0G5_20261015-1432.cif" ] || { echo "ERROR: mdbridge dict finish did not write golden_03_0G5_20261015-1432.cif" >&2; exit 1; }

echo "Restraint dictionaries: golden_03_0C3_20261015-1432.cif golden_03_0G5_20261015-1432.cif"
