#!/bin/bash
# GROMACS: conversion and MD stages
# Project apt1, step 09 - written by MDBridge <VERSION> on <DATE>.
# MDBridge only writes this script; review it, then run it with: bash apt1_09_gromacs_20261015-1432.sh
# GROMACS 2024+.
# Please cite (GROMACS, ParmEd):
#   - Abraham M. J., Murtola T., Schulz R., Páll S., Smith J. C., Hess B., Lindahl E. SoftwareX
#     2015, 1-2, 19-25. doi:10.1016/j.softx.2015.06.001
#   - Shirts M. R., Klein C., Swails J. M., Yin J., Gilson M. K., Mobley D. L., Case D. A., Zhong E.
#     D. J. Comput.-Aided Mol. Des. 2017, 31, 147-161. doi:10.1007/s10822-016-9977-1

set -eo pipefail
cd "$(dirname "$0")"

GMX="gmx"

# Convert the Amber topology / coordinates with ParmEd
python3 apt1_09_convert_20261015-1432.py ../03_amber-build/apt1_03_parmed_20261015-1432.parm7 ../03_amber-build/apt1_03_tleap_20261015-1432.rst7 apt1_09_convert_20261015-1432.top apt1_09_convert_20261015-1432.gro
[ -s "apt1_09_convert_20261015-1432.top" ] || { echo "ERROR: ParmEd did not write apt1_09_convert_20261015-1432.top" >&2; exit 1; }

# Position restraints for the RNA (used by stages with define = -DPOSRES)
python3 apt1_09_posre_20261015-1432.py apt1_09_convert_20261015-1432.top

# em
$GMX grompp -f apt1_09_em_20261015-1432.mdp -c apt1_09_convert_20261015-1432.gro -p apt1_09_convert_20261015-1432.top -o apt1_09_em_20261015-1432.tpr -maxwarn 0
$GMX mdrun -deffnm apt1_09_em_20261015-1432 -nb gpu -pme gpu -bonded gpu
[ -s "apt1_09_em_20261015-1432.gro" ] || { echo "ERROR: mdrun (em) did not write apt1_09_em_20261015-1432.gro" >&2; exit 1; }

# nvt
$GMX grompp -f apt1_09_nvt_20261015-1432.mdp -c apt1_09_em_20261015-1432.gro -r apt1_09_em_20261015-1432.gro -p apt1_09_convert_20261015-1432.top -o apt1_09_nvt_20261015-1432.tpr -maxwarn 0
$GMX mdrun -deffnm apt1_09_nvt_20261015-1432 -nb gpu -pme gpu -bonded gpu
[ -s "apt1_09_nvt_20261015-1432.gro" ] || { echo "ERROR: mdrun (nvt) did not write apt1_09_nvt_20261015-1432.gro" >&2; exit 1; }

# npt
$GMX grompp -f apt1_09_npt_20261015-1432.mdp -c apt1_09_nvt_20261015-1432.gro -r apt1_09_nvt_20261015-1432.gro -t apt1_09_nvt_20261015-1432.cpt -p apt1_09_convert_20261015-1432.top -o apt1_09_npt_20261015-1432.tpr -maxwarn 0
$GMX mdrun -deffnm apt1_09_npt_20261015-1432 -nb gpu -pme gpu -bonded gpu
[ -s "apt1_09_npt_20261015-1432.gro" ] || { echo "ERROR: mdrun (npt) did not write apt1_09_npt_20261015-1432.gro" >&2; exit 1; }
