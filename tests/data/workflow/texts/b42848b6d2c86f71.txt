#!/bin/bash
# MDBridge structure builder: Ideal A-form helix
# Project golden, step 03 - written by MDBridge <VERSION> on <DATE>.
# MDBridge only writes this script; review it, then run it with: bash golden_03_build_20261015-1432.sh
# MDBridge must be on PATH, or MDBRIDGE must name the program.
# Please cite (Arnott A-RNA fiber model (MDATK helix builder)):
#   - Arnott S., Hukins D. W. L., Dover S. D. Biochem. Biophys. Res. Commun. 1972, 48, 1392-1399.
#     doi:10.1016/0006-291X(72)90867-4

set -eo pipefail
cd "$(dirname "$0")"

# MDBridge: the mdbridge program on PATH, unless MDBRIDGE names another
MDBRIDGE="${MDBRIDGE:-mdbridge}"

"$MDBRIDGE" build helix --seq '' --partner auto --chirality D --chains A,B --start=1,1 --scheme pdb -o golden_03_helix_20261015-1432.pdb
[ -s "golden_03_helix_20261015-1432.pdb" ] || { echo "ERROR: mdbridge build did not write golden_03_helix_20261015-1432.pdb" >&2; exit 1; }
