# Solvate and add ions (OL3+CP) for project golden, step 01
# written by MDBridge <VERSION>
source leaprc.RNA.OL3
source leaprc.water.spce
addAtomTypes { { "LD" "O" "sp3" } { "LH" "O" "sp3" } }
addAtomTypes { { "OR" "O" "sp3" } }
loadamberparams frcmod.cp
loadOff RNA_CP.lib
mol = loadpdb "golden_01_lrna.pdb"
solvateoct mol SPCBOX 15.0

addIonsRand mol Na+ 0

check mol
saveamberparm mol golden_01_tleap.parm7 golden_01_tleap.rst7
savepdb mol golden_01_tleap.pdb
quit
