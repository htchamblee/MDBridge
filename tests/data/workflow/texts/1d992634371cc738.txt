set -eo pipefail
cd "$(dirname "$0")"

PMEMD="mpirun -np 8 pmemd.MPI"
TOP="/golden/inputs/topology.parm7"

# a stage counts as finished when its output has the final timing lines
# (pmemd: "Total wall time", sander: "Run   done at")
finished() { [ -s "$1" ] && grep -qE "Total wall time|Run +done at" "$1"; }

# prod: Production, helix pairs held
if finished golden_03_prod_20261015-1432.out; then
  echo "prod: already finished - skipping"
else
  $PMEMD -O -i golden_03_prod_20261015-1432.in -p "$TOP" -c golden_03_eq-free_20261015-1432.rst7 -ref golden_03_eq-free_20261015-1432.rst7 -o golden_03_prod_20261015-1432.out -r golden_03_prod_20261015-1432.rst7 -x golden_03_prod_20261015-1432.nc -inf golden_03_prod_20261015-1432.mdinfo
fi
[ -s "golden_03_prod_20261015-1432.rst7" ] || { echo "ERROR: pmemd (prod) did not write golden_03_prod_20261015-1432.rst7" >&2; exit 1; }
