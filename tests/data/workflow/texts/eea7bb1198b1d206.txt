# Solvate and add ions (OL3+CP) for project golden, step 03
# written by MDBridge <VERSION>
source leaprc.RNA.OL3
source leaprc.water.tip4pew
loadamberparams frcmod.tip4p2005
loadOff tip4p2005box.off
WAT = TIP4P2005
HOH = TIP4P2005
addAtomTypes { { "LD" "O" "sp3" } { "LH" "O" "sp3" } }
addAtomTypes { { "OR" "O" "sp3" } }
loadamberparams frcmod.cp
loadOff RNA_CP.lib
loadamberparams golden_03_mg_20261015-1432.frcmod
mol = loadpdb "golden_03_lrna_20261015-1432.pdb"
solvateoct mol TIP4P2005BOX 15.0

addIonsRand mol Na+ 0
addIonsRand mol MG 4
addIonsRand mol Cl- 8

check mol
saveamberparm mol golden_03_tleap_20261015-1432.parm7 golden_03_tleap_20261015-1432.rst7
savepdb mol golden_03_tleap_20261015-1432.pdb
quit
