#     6183-6191. doi:10.1021/acs.jcim.3c01153

#     2013, 9, 3878-3888. doi:10.1021/ct400314y

set -eo pipefail
cd "$(dirname "$0")"

PMEMD="pmemd.cuda"
TOP="../03_amber-build/apt1_03_parmed_20261015-1432.parm7"

# a stage counts as finished when its output has the final timing lines
# (pmemd: "Total wall time", sander: "Run   done at")
finished() { [ -s "$1" ] && grep -qE "Total wall time|Run +done at" "$1"; }

# prod: Production, helix pairs held
if finished apt1_05_prod_20261015-1432.out; then
  echo "prod: already finished - skipping"
else
  $PMEMD -O -i apt1_05_prod_20261015-1432.in -p "$TOP" -c apt1_05_eq-free_20261015-1432.rst7 -ref apt1_05_eq-free_20261015-1432.rst7 -o apt1_05_prod_20261015-1432.out -r apt1_05_prod_20261015-1432.rst7 -x apt1_05_prod_20261015-1432.nc -inf apt1_05_prod_20261015-1432.mdinfo
fi
[ -s "apt1_05_prod_20261015-1432.rst7" ] || { echo "ERROR: pmemd (prod) did not write apt1_05_prod_20261015-1432.rst7" >&2; exit 1; }

# Dry, imaged copy of the production trajectory (and its topology)
cpptraj << EOF
parm "$TOP"
trajin apt1_05_prod_20261015-1432.nc
autoimage
strip :WAT,HOH,Na+,Cl-,K+,MG parmout apt1_05_dry_20261015-1432.parm7
trajout apt1_05_dry-prod_20261015-1432.nc netcdf
run
EOF
[ -s "apt1_05_dry-prod_20261015-1432.nc" ] || { echo "ERROR: cpptraj did not write apt1_05_dry-prod_20261015-1432.nc" >&2; exit 1; }
