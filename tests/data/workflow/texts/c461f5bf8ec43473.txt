source leaprc.RNA.OL3
source leaprc.water.spce
addAtomTypes { { "LD" "O" "sp3" } { "LH" "O" "sp3" } }

loadamberparams frcmod.cp

saveamberparm mol apt1_03_tleap_20261015-1432.parm7 apt1_03_tleap_20261015-1432.rst7
savepdb mol apt1_03_tleap_20261015-1432.pdb
quit
